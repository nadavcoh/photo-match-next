-- =============================================================================
-- Supabase Auth – user creation restriction (optional extra safety net)
-- =============================================================================
--
-- The restriction that is actually in use is the "Before User Created" auth hook
-- public.restrict_signup (supabase/migrations/20261117130000_restrict_signup_by_github_id.sql)
-- plus the middleware check (src/middleware.ts). This script is NOT installed on
-- the project; it is a second, independent layer you can add in the SQL Editor.
--
-- What it does
-- ────────────
-- Creates a BEFORE INSERT trigger on auth.users that raises an exception (and
-- therefore rolls back the insert) unless the new user signed up with GitHub and
-- the GitHub user id is the owner's (29537286).
--
-- raw_app_meta_data->>'provider' is set by the auth server. The GitHub id is read
-- from raw_user_meta_data->>'provider_id', which GitHub sign-ups carry; the
-- provider check is what stops an email sign-up from simply claiming that id.
-- =============================================================================


-- ── 1. Trigger function ───────────────────────────────────────────────────────

CREATE OR REPLACE FUNCTION auth.restrict_user_creation()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = auth   -- pin search_path to prevent search-path-hijacking
AS $$
BEGIN
  IF NEW.raw_app_meta_data ->> 'provider' IS DISTINCT FROM 'github'
     OR NEW.raw_user_meta_data ->> 'provider_id' IS DISTINCT FROM '29537286' THEN
    RAISE EXCEPTION 'Registration is not permitted. This application is private.';
  END IF;

  RETURN NEW;  -- allow the insert to proceed
END;
$$;


-- ── 2. Attach the trigger ─────────────────────────────────────────────────────

-- Drop first so re-running this script is idempotent.
DROP TRIGGER IF EXISTS restrict_user_creation_trigger ON auth.users;

CREATE TRIGGER restrict_user_creation_trigger
  BEFORE INSERT ON auth.users
  FOR EACH ROW
  EXECUTE FUNCTION auth.restrict_user_creation();


-- ── Done ──────────────────────────────────────────────────────────────────────
